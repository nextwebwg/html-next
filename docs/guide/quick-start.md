# Build an HTML Next component

Write a workshop check-in counter once in HTML, then use it natively or in Vue, React, or Svelte.

## Write the component

Save this as `counter.html`: each click counts a guest, and Reset starts the next session.

```html title="counter.html"
<template component="x-counter">
  <defs>
    <state name="count" type="number" value="0"></state>
    <handler name="increment">
      <set name="count" expr:value="$count + 1"></set>
    </handler>
    <handler name="reset">
      <set name="count" expr:value="0"></set>
    </handler>
  </defs>
  <div>
    <button type="button" on:click="increment">
      Count: {$count}
    </button>
    <button type="button" on:click="reset">Reset</button>
  </div>
</template>
```

::: framework-html-next

Nothing to convert: the browser runtime loads `counter.html` as written, or Vite builds it into native DOM factories. [Set up HTML Next](/html-next/usage).

:::

::: framework-vue

The Vue adapter generates this component when your app builds. [Set up Vue](/html-next/usage/vue).

```vue title="XCounter.vue"
<!-- Generated by HTML Next 1.0.0-alpha.41 for Vue 3.5. Do not edit. -->
<script setup lang="ts">
import { provide, ref } from 'vue'
import { acceptsWrite, checkHydratedRoot, isNumber, nativeAttrs } from './host'

defineOptions({ inheritAttrs: false })

const count = ref(0)
provide('html-next:x-counter:count', count)
checkHydratedRoot('div', 'x-counter')

function increment(): void {
  const next = count.value + 1
  if (acceptsWrite(next, isNumber, 'counter.html', 'increment', 'count')) count.value = next
}

function reset(): void {
  count.value = 0
}
</script>

<template>
  <div v-bind="nativeAttrs($attrs)">
    <button type="button" @click="increment">Count: {{ count }}</button>
    <button type="button" @click="reset">Reset</button>
  </div>
</template>
```

::: framework-react

The React adapter generates this component when your app builds. [Set up React](/html-next/usage/react).

```tsx title="XCounter.tsx"
// Generated by HTML Next 1.0.0-alpha.41 for React 19. Do not edit.
import React from "react";
import type { ReactNode } from "react";
import { componentContext } from "./context";
import { useNativeEvents } from "./events";
import { acceptsWrite, isNumber, text, useLiveState } from "./render";
export const HtmlNextContext_XCounter_count = componentContext<number>("x-counter", "count");

export type XCounterProps = Omit<React.HTMLAttributes<HTMLDivElement>, "style"> & {
  style?:
    | React.CSSProperties
    | (React.CSSProperties & { [name: `--${string}`]: string | number | undefined });
  children?: ReactNode;
  slots?: Readonly<Record<string, ReactNode | ((props: Record<string, any>) => ReactNode)>>;
  ref?: React.Ref<HTMLDivElement>;
  [name: string]: unknown;
};

export default function XCounter(props: XCounterProps) {
  const { children: _children, slots: _slots, ...nativeAttrs } = props;
  const [count, setCount, getCount] = useLiveState<number>(0);
  const rootRef = React.useRef<HTMLDivElement | null>(null);
  React.useImperativeHandle(props.ref, () => rootRef.current!, undefined);
  function increment(): void {
    const next = getCount() + 1;
    if (acceptsWrite(next, isNumber, "counter.html", "increment", "count")) setCount(next);
  }
  function reset(): void {
    setCount(0);
  }
  const incrementEvents = useNativeEvents([{ type: "click", modifiers: [], handler: increment }]);
  const resetEvents = useNativeEvents([{ type: "click", modifiers: [], handler: reset }]);
  return (
    <HtmlNextContext_XCounter_count value={{ value: count }}>
      <div
        {...nativeAttrs}
        ref={rootRef}
        data-component={[nativeAttrs["data-component"], "x-counter"].filter(Boolean).join(" ")}
      >
        <button type="button" ref={incrementEvents}>{`Count: ${text(count)}`}</button>
        <button type="button" ref={resetEvents}>Reset</button>
      </div>
    </HtmlNextContext_XCounter_count>
  );
}
```
